from pathlib import Path
import sys,json,math
import numpy as np
from scipy.special import kve
P=Path(__file__).resolve().parent;A=P.parent/'production_lepton_port';sys.path.insert(0,str(A/'test_mirror/src'));sys.dont_write_bytecode=True
from nmbackend.reference import finite_t_leptons as L
from nmbackend.constants import HBARC,M_E
rows=[]
for z in [40.,100.,300.,600.]:
 T=M_E/z;s=L.state_mu(T,0.);want=M_E*M_E*T/(math.pi**2*HBARC**3)*kve(2,z)*math.exp(-z)
 # Fermi correction relative <=exp(-z), negligible at declared 3e-8 gate.
 error=s.particle_density/want-1
 fine=L.state_mu(T,0.,tail=90.,rtol=4e-13)
 rows.append(dict(z=z,T=T,particle=s.particle_density,bessel=want,relative_error=error,gate=3e-8,passed=abs(error)<=3e-8,tail_fine_relative=fine.particle_density/s.particle_density-1,susceptibility=s.dn_dmu_fixed_temperature))
out=dict(status='FRESH_DILUTE_PHYSICAL_FORWARD_PROBE',rows=rows,passed=sum(r['passed']for r in rows),failed=[r for r in rows if not r['passed']]);(P/'dilute_results.json').write_text(json.dumps(out,indent=2)+'\n');print(json.dumps(out,indent=2))
